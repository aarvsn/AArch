     1→/*
     2→ * mgbax CPU: ARM7TDMI (ARMv4T) — ARM and Thumb instruction sets.
     3→ *
     4→ * Spec-driven details implemented:
     5→ *  - full condition-code table; conditional execution of every instruction
     6→ *  - barrel shifter with carry-out semantics (LSL/LSR/ASR/ROR/RRX, register
     7→ *    and immediate shifts, shift-by-32+ values, ROR-by-0 = RRX)
     8→ *  - ADC/SBC carry-in semantics (C = !borrow for SBC), CMN/CMP/TST/TEQ
     9→ *  - PC pipeline reads: +8 ARM, +4 Thumb; PC as Rm with register-specified
    10→ *    shift reads +12 (documented ARM7TDMI quirk)
    11→ *  - MUL/MLA/UMULL/UMLAL/SMULL/SMLAL (S variants leave C unchanged: the
    12→ *    ARMv4 "undefined" choice, documented)
    13→ *  - SWP/SWPB, halfword transfers (LDRH/STRH/LDRSB/LDRSH), LDR/STR with
    14→ *    pre/post-index + writeback, unaligned word loads rotate right
    15→ *  - LDM/STM (IA/IB/DA/DB), writeback, PC write semantics
    16→ *  - BX interworking, MRS/MSR with field masks and bank switching,
    17→ *    banked registers incl. FIQ r8-r12
    18→ *  - SWI routed to the HLE BIOS layer; exceptions via banked modes
    19→ *
    20→ * Cycle counts: flat deterministic model documented per class (not
    21→ * per-bus-cycle accurate).
    22→ */
    23→#include "gba.h"
    24→
    25→#include <string.h>

    27→/* ---- banked registers ------------------------------------------------------- */
    28→
    29→static int bank_index(uint8_t mode)
    30→{
    31→    switch (mode) {
    32→    case GBA_MODE_FIQ: return 1;
    33→    case GBA_MODE_IRQ: return 2;
    34→    case GBA_MODE_SVC: return 3;
    35→    case GBA_MODE_ABT: return 4;
    36→    case GBA_MODE_UND: return 5;
    37→    default: return 0; /* usr/sys */
    38→    }
    39→}
    40→
    41→void gba_cpu_set_mode(gba_cpu *c, uint8_t new_mode)
    42→{
    43→    int old_b = bank_index(c->mode);
    44→    int new_b = bank_index(new_mode);
    45→    c->mode = new_mode;
    46→    c->cpsr = (c->cpsr & ~0x1Fu) | (new_mode & 0x1Fu);
    47→    if (old_b == new_b)
    48→        return;
    49→    c->bank_r13[old_b] = c->r[13];
    50→    c->bank_r14[old_b] = c->r[14];
    51→    c->spsr[old_b] = c->spsr[6]; /* current SPSR cache */
    52→    if (old_b == 1)
    53→        memcpy(c->bank_r8_fiq, &c->r[8], 5 * sizeof(uint32_t));
    54→    c->r[13] = c->bank_r13[new_b];
    55→    c->r[14] = c->bank_r14[new_b];
    56→    c->spsr[6] = c->spsr[new_b];
    57→    if (new_b == 1)
    58→        memcpy(&c->r[8], c->bank_r8_fiq, 5 * sizeof(uint32_t));
    59→}
    60→
    61→uint32_t gba_cpu_read_spsr(const gba_cpu *c)
    62→{
    63→    return c->spsr[6]; /* user mode: SPSR unavailable; reads cached value */
    64→}
    65→
    66→void gba_cpu_write_spsr(gba_cpu *c, uint32_t v)
    67→{
    68→    c->spsr[6] = v;
    69→}
    70→
    71→void gba_cpu_exception(gba_t *g, uint32_t vector, uint8_t new_mode,
    72→                       uint32_t lr_offset, int save_spsr)
    73→{
    74→    gba_cpu *c = &g->cpu;
    75→    uint32_t old_cpsr = c->cpsr;
    76→    uint32_t lr = c->r[15] + lr_offset;
    77→    gba_cpu_set_mode(c, new_mode);
    78→    if (save_spsr)
    79→        c->spsr[6] = old_cpsr;
    80→    c->r[14] = lr;
    81→    c->cpsr |= GBA_I;
    82→    c->cpsr &= (uint32_t)~GBA_T;
    83→    c->r[15] = vector & ~3u;
    84→}

    86→void gba_cpu_power_on(gba_cpu *c)
    87→{
    88→    memset(c, 0, sizeof *c);
    89→    c->mode = GBA_MODE_SVC;
    90→    c->cpsr = GBA_I | GBA_F | GBA_MODE_SVC;
    91→    c->r[13] = 0x03007F00u;
    92→    c->r[15] = 0x08000000u;
    93→}

    96→
    97→static int cond_ok(uint32_t cond, uint32_t cpsr)
    98→{
    99→    uint32_t n = cpsr & GBA_N, z = cpsr & GBA_Z, cc = cpsr & GBA_C, v = cpsr & GBA_V;
   100→    switch (cond & 0xFu) {
   101→    case 0x0: return z != 0;
   102→    case 0x1: return z == 0;
   103→    case 0x2: return cc != 0;
   104→    case 0x3: return cc == 0;
   105→    case 0x4: return n != 0;
   106→    case 0x5: return n == 0;
   107→    case 0x6: return v != 0;
   108→    case 0x7: return v == 0;
   109→    case 0x8: return cc != 0 && z == 0;
   110→    case 0x9: return cc == 0 || z != 0;
   111→    case 0xA: return n == v;
   112→    case 0xB: return n != v;
   113→    case 0xC: return z == 0 && n == v;
   114→    case 0xD: return z != 0 || n != v;
   115→    default: return 1; /* AL (and NV treated as AL for milestone) */
   116→    }
   117→}

   119→/* ---- barrel shifter -------------------------------------------------------------- */
   120→
   121→typedef struct {
   122→    uint32_t value;
   123→    uint8_t carry_out;
   124→} shift_result;
   125→
   126→static shift_result shift_calc(uint32_t value, uint8_t type, uint32_t amount,
   127→                               uint8_t carry_in)
   128→{
   129→    shift_result r;
   130→    r.value = value;
   131→    r.carry_out = carry_in;
   132→    switch (type) {
   133→    case 0: /* LSL */
   134→        if (amount == 0u)
   135→            break;
   136→        if (amount < 32u) {
   137→            r.carry_out = (uint8_t)((value >> (32u - amount)) & 1u);
   138→            r.value = value << amount;
   139→        } else if (amount == 32u) {
   140→            r.carry_out = (uint8_t)(value & 1u);
   141→            r.value = 0;
   142→        } else {
   143→            r.carry_out = 0;
   144→            r.value = 0;
   145→        }
   146→        break;
   147→    case 1: /* LSR */
   148→        if (amount == 0u)
   149→            amount = 32; /* register form LSR #0 */
   150→        if (amount < 32u) {
   151→            r.carry_out = (uint8_t)((value >> (amount - 1u)) & 1u);
   152→            r.value = value >> amount;
   153→        } else if (amount == 32u) {
   154→            r.carry_out = (uint8_t)((value >> 31) & 1u);
   155→            r.value = 0;
   156→        } else {
   157→            r.carry_out = 0;
   158→            r.value = 0;
   159→        }
   160→        break;
   161→    case 2: /* ASR */
   162→        if (amount == 0u || amount >= 32u) {
   163→            r.carry_out = (uint8_t)((value >> 31) & 1u);
   164→            r.value = (value & 0x80000000u) ? 0xFFFFFFFFu : 0u;
   165→            break;
   166→        }
   167→        r.carry_out = (uint8_t)((value >> (amount - 1u)) & 1u);
   168→        r.value = (uint32_t)((int32_t)value >> amount);
   169→        break;
   170→    default: /* ROR */
   171→        if (amount == 0u) { /* RRX */
   172→            r.carry_out = (uint8_t)(value & 1u);
   173→            r.value = (value >> 1) | (carry_in ? 0x80000000u : 0u);
   174→            break;
   175→        }
   176→        amount &= 31u;
   177→        if (amount == 0u) {
   178→            r.carry_out = (uint8_t)((value >> 31) & 1u);
   179→            r.value = value;
   180→        } else {
   181→            r.carry_out = (uint8_t)((value >> (amount - 1u)) & 1u);
   182→            r.value = (value >> amount) | (value << (32u - amount));
   183→        }
   184→        break;
   185→    }
   186→    return r;
   187→}

   189→static inline uint8_t cpsr_c(const gba_cpu *c)
   190→{
   191→    return (uint8_t)((c->cpsr & GBA_C) ? 1 : 0);
   192→}

   194→/* decode ARM operand2 (register form); pc_offset adds the pipeline offset
   195→ * for PC-relative reads (0 for immediate shifts, +4 for register shifts) */
   196→static shift_result arm_operand2(gba_cpu *c, uint32_t instr)
   197→{
   198→    uint8_t type = (uint8_t)((instr >> 5) & 3u);
   199→    uint8_t rm = (uint8_t)(instr & 0xFu);
   200→    if (instr & (1u << 4)) { /* register-specified shift */
   201→        uint32_t amount = c->r[(instr >> 8) & 0xFu] & 0xFFu;
   202→        uint32_t value = (rm == 15u) ? (c->r[15] + 4u) : c->r[rm];
   203→        if (amount == 0u) { /* shift by 0: unchanged, C unaffected */
   204→            shift_result r;
   205→            r.value = value;
   206→            r.carry_out = cpsr_c(c);
   207→            return r;
   208→        }
   209→        return shift_calc(value, type, amount, cpsr_c(c));
   210→    }
   211→    uint32_t amount = (instr >> 7) & 0x1Fu;
   212→    uint32_t value = (rm == 15u) ? c->r[15] : c->r[rm];
   213→    if (amount != 0u)
   214→        return shift_calc(value, type, amount, cpsr_c(c));
   215→    /* shift-by-zero immediates */
   216→    shift_result r;
   217→    if (type == 0u) {
   218→        r.value = value;
   219→        r.carry_out = cpsr_c(c);
   220→    } else if (type == 3u) { /* ROR #0 = RRX */
   221→        r.carry_out = (uint8_t)(value & 1u);
   222→        r.value = (value >> 1) | (cpsr_c(c) ? 0x80000000u : 0u);
   223→    } else if (type == 1u) { /* LSR #32 */
   224→        r.carry_out = (uint8_t)((value >> 31) & 1u);
   225→        r.value = 0;
   226→    } else { /* ASR #32 */
   227→        r.carry_out = (uint8_t)((value >> 31) & 1u);
   228→        r.value = (value & 0x80000000u) ? 0xFFFFFFFFu : 0u;
   229→    }
   230→    return r;
   231→}
   232→
   233→/* immediate operand2 with rotation; carry = bit rotated out */
   234→static uint32_t arm_imm(gba_cpu *c, uint32_t instr, uint8_t *carry_out)
   235→{
   236→    uint32_t imm = instr & 0xFFu;
   237→    uint32_t rot = ((instr >> 8) & 0xFu) * 2u;
   238→    if (rot == 0u) {
   239→        *carry_out = cpsr_c(c);
   240→        return imm;
   241→    }
   242→    uint32_t v = (imm >> rot) | (imm << (32u - rot));
   243→    *carry_out = (uint8_t)((v >> 31) & 1u);
   244→    return v;
   245→}

   247→/* ---- ALU ------------------------------------------------------------------------ */
   248→
   249→static inline void set_nz32(gba_cpu *c, uint32_t v)
   250→{
   251→    c->cpsr &= (uint32_t)~(GBA_N | GBA_Z);
   252→    if (v == 0u)
   253→        c->cpsr |= GBA_Z;
   254→    if (v & 0x80000000u)
   255→        c->cpsr |= GBA_N;
   256→}
   257→
   258→static inline uint32_t add_with_carry(gba_cpu *c, uint32_t a, uint32_t b,
   259→                                      uint32_t carry_in, int set_flags)
   260→{
   261→    uint64_t sum = (uint64_t)a + b + carry_in;
   262→    uint32_t r = (uint32_t)sum;
   263→    if (set_flags) {
   264→        c->cpsr &= (uint32_t)~(GBA_N | GBA_Z | GBA_C | GBA_V);
   265→        if (sum > 0xFFFFFFFFu)
   266→            c->cpsr |= GBA_C;
   267→        if (~(a ^ b) & (a ^ r) & 0x80000000u)
   268→            c->cpsr |= GBA_V;
   269→        set_nz32(c, r);
   270→    }
   271→    return r;
   272→}
   273→
   274→static inline uint32_t sub_with_carry(gba_cpu *c, uint32_t a, uint32_t b,
   275→                                      uint32_t borrow_in, int set_flags)
   276→{
   277→    uint32_t r = a - b - borrow_in;
   278→    if (set_flags) {
   279→        c->cpsr &= (uint32_t)~(GBA_N | GBA_Z | GBA_C | GBA_V);
   280→        if (a >= b + borrow_in)
   281→            c->cpsr |= GBA_C;
   282→        if ((a ^ b) & (a ^ r) & 0x80000000u)
   283→            c->cpsr |= GBA_V;
   284→        set_nz32(c, r);
   285→    }
   286→    return r;
   287→}
   288→
   289→static inline uint32_t get_reg(gba_cpu *c, uint8_t r, uint32_t extra_pc_offset)
   290→{
   291→    return (r == 15u) ? (c->r[15] + extra_pc_offset) : c->r[r];
   292→}
   293→
   294→static void do_msr(gba_cpu *c, uint32_t instr, uint32_t value)
   295→{
   296→    int use_spsr = (instr & (1u << 22)) != 0;
   297→    uint32_t mask = 0;
   298→    if (instr & (1u << 19)) mask |= 0xFF000000u;
   299→    if (instr & (1u << 18)) mask |= 0x00FF0000u;
   300→    if (instr & (1u << 17)) mask |= 0x0000FF00u;
   301→    if (instr & (1u << 16)) mask |= 0x000000FFu;
   302→    if (use_spsr) {
   303→        uint32_t spsr = gba_cpu_read_spsr(c);
   304→        gba_cpu_write_spsr(c, (spsr & ~mask) | (value & mask));
   305→        return;
   306→    }
   307→    if (c->mode == GBA_MODE_USR)
   308→        mask &= 0xFF000000u; /* user mode: flags only */
   309→    uint32_t new_cpsr = (c->cpsr & ~mask) | (value & mask);
   310→    uint8_t new_mode = (uint8_t)(new_cpsr & 0x1Fu);
   311→    c->cpsr = new_cpsr;
   312→    gba_cpu_set_mode(c, new_mode);
   313→}

   315→/* ---- ARM step ---------------------------------------------------------------------- */
   316→
   317→static uint32_t arm_step(gba_t *g, gba_cpu *c, uint32_t instr)
   318→{
   319→    if (!cond_ok(instr >> 28, c->cpsr)) {
   320→        c->r[15] += 4u;
   321→        return 1;
   322→    }
   323→
   324→    /* SWI: HLE */
   325→    if ((instr & 0x0F000000u) == 0x0F000000u) {
   326→        uint32_t comment = instr & 0xFFu;
   327→        gba_swi_hle(g, comment);
   328→        c->r[15] += 4u;
   329→        return 3;
   330→    }
   331→
   332→    /* BX */
   333→    if ((instr & 0x0FFFFFF0u) == 0x012FFF10u) {
   334→        uint32_t target = c->r[instr & 0xFu];
   335→        c->cpsr = (c->cpsr & (uint32_t)~GBA_T) | (target & 1u ? GBA_T : 0u);
   336→        c->r[15] = target & ~1u;
   337→        return 3;
   338→    }
   339→
   340→    /* multiply family */
   341→    if ((instr & 0x0FC000F0u) == 0x00000090u) {
   342→        uint8_t rd = (uint8_t)((instr >> 16) & 0xFu);
   343→        uint8_t rn = (uint8_t)((instr >> 12) & 0xFu);
   344→        uint8_t rs = (uint8_t)((instr >> 8) & 0xFu);
   345→        uint8_t rm = (uint8_t)(instr & 0xFu);
   346→        int set_flags = (instr & (1u << 20)) != 0;
   347→        int accum = (instr & (1u << 21)) != 0;
   348→        int is64 = (instr & (1u << 23)) != 0;
   349→        int signed_mul = (instr & (1u << 22)) != 0;
   350→        uint32_t m = c->r[rm], sv = c->r[rs];
   351→        if (is64) {
   352→            uint64_t result;
   353→            if (signed_mul) {
   354→                int64_t sm = (int64_t)(int32_t)m * (int64_t)(int32_t)sv;
   355→                if (accum)
   356→                    sm += (int64_t)((uint64_t)c->r[rn] | ((uint64_t)c->r[rd] << 32));
   357→                result = (uint64_t)sm;
   358→            } else {
   359→                uint64_t um = (uint64_t)m * sv;
   360→                if (accum)
   361→                    um += (uint64_t)c->r[rn] | ((uint64_t)c->r[rd] << 32);
   362→                result = um;
   363→            }
   364→            c->r[rn] = (uint32_t)result;
   365→            c->r[rd] = (uint32_t)(result >> 32);
   366→            if (set_flags) {
   367→                c->cpsr &= (uint32_t)~(GBA_N | GBA_Z);
   368→                if (result == 0u)
   369→                    c->cpsr |= GBA_Z;
   370→                if (result & 0x8000000000000000ull)
   371→                    c->cpsr |= GBA_N;
   372→            }
   373→        } else {
   374→            uint32_t result = accum ? m * sv + c->r[rn] : m * sv;
   375→            c->r[rd] = result;
   376→            if (set_flags)
   377→                set_nz32(c, result);
   378→        }
   379→        c->r[15] += 4u;
   380→        return 4;
   381→    }
   382→
   383→    /* MRS */
   384→    if ((instr & 0x0FBF0FFFu) == 0x010F0000u) {
   385→        uint8_t rd = (uint8_t)((instr >> 12) & 0xFu);
   386→        c->r[rd] = (instr & (1u << 22)) ? gba_cpu_read_spsr(c) : c->cpsr;
   387→        c->r[15] += 4u;
   388→        return 2;
   389→    }
   390→
   391→    /* MSR (immediate and register) */
   392→    if ((instr & 0x0DB0F000u) == 0x0120F000u) {
   393→        uint32_t value;
   394→        if (instr & (1u << 25)) {
   395→            uint8_t carry;
   396→            value = arm_imm(c, instr, &carry);
   397→        } else {
   398→            value = c->r[instr & 0xFu];
   399→        }
   400→        do_msr(c, instr, value);
   401→        c->r[15] += 4u;
   402→        return 2;
   403→    }
   404→
   405→    /* SWP/SWPB */
   406→    if ((instr & 0x0FB00FF0u) == 0x01000090u) {
   407→        uint8_t rn = (uint8_t)((instr >> 16) & 0xFu);
   408→        uint8_t rd = (uint8_t)((instr >> 12) & 0xFu);
   409→        uint8_t rm = (uint8_t)(instr & 0xFu);
   410→        uint32_t addr = c->r[rn];
   411→        if (instr & (1u << 22)) {
   412→            uint8_t tmp = gba_mem_read8(g, addr);
   413→            gba_mem_write8(g, addr, (uint8_t)(c->r[rm] & 0xFFu));
   414→            c->r[rd] = tmp;
   415→        } else {
   416→            uint32_t tmp = gba_bus_read32(g, addr);
   417→            gba_mem_write32(g, addr, c->r[rm]);
   418→            c->r[rd] = tmp;
   419→        }
   420→        c->r[15] += 4u;
   421→        return 4;
   422→    }
   423→
   424→    /* halfword / signed byte transfers (bits 4 and 7 set) */
   425→    if ((instr & 0x0E000090u) == 0x00000090u && (instr & 0x60u) != 0u) {
   426→        uint8_t rn = (uint8_t)((instr >> 16) & 0xFu);
   427→        uint8_t rd = (uint8_t)((instr >> 12) & 0xFu);
   428→        int up = (instr & (1u << 23)) != 0;
   429→        int pre = (instr & (1u << 24)) != 0;
   430→        int writeback = (instr & (1u << 21)) != 0;
   431→        int load = (instr & (1u << 20)) != 0;
   432→        uint32_t offset = (instr & (1u << 22))
   433→                              ? ((uint32_t)(instr >> 4) & 0xFu) |
   434→                                    ((uint32_t)(instr >> 8) & 0xF0u)
   435→                              : c->r[instr & 0xFu];
   436→        uint32_t base = get_reg(c, rn, 0);
   437→        uint32_t addr = up ? base + offset : base - offset;
   438→        uint8_t kind = (uint8_t)((instr >> 5) & 3u);
   439→        if (pre) {
   440→            if (!load) {
   441→                if (kind == 1u)
   442→                    gba_mem_write16(g, addr, (uint16_t)c->r[rd]);
   443→            } else if (kind == 1u) {
   444→                c->r[rd] = gba_bus_read16(g, addr);
   445→            } else if (kind == 2u) {
   446→                c->r[rd] = (uint32_t)(int32_t)(int8_t)gba_mem_read8(g, addr);
   447→            } else {
   448→                c->r[rd] = (uint32_t)(int32_t)(int16_t)gba_bus_read16(g, addr);
   449→            }
   450→            if (writeback && (!load || rd != rn))
   451→                c->r[rn] = addr;
   452→        } else {
   453→            if (!load) {
   454→                if (kind == 1u)
   455→                    gba_mem_write16(g, base, (uint16_t)c->r[rd]);
   456→            } else if (kind == 1u) {
   457→                c->r[rd] = gba_bus_read16(g, base);
   458→            } else if (kind == 2u) {
   459→                c->r[rd] = (uint32_t)(int32_t)(int8_t)gba_mem_read8(g, base);
   460→            } else {
   461→                c->r[rd] = (uint32_t)(int32_t)(int16_t)gba_bus_read16(g, base);
   462→            }
   463→            c->r[rn] = addr;
   464→        }
   465→        c->r[15] += 4u;
   466→        return load ? 3u : 2u;
   467→    }
   468→
   469→    /* single data transfer */
   470→    if ((instr & 0x0C000000u) == 0x04000000u) {
   471→        uint8_t rn = (uint8_t)((instr >> 16) & 0xFu);
   472→        uint8_t rd = (uint8_t)((instr >> 12) & 0xFu);
   473→        int up = (instr & (1u << 23)) != 0;
   474→        int pre = (instr & (1u << 24)) != 0;
   475→        int writeback = (instr & (1u << 21)) != 0;
   476→        int load = (instr & (1u << 20)) != 0;
   477→        int byte = (instr & (1u << 22)) != 0;
   478→        uint32_t base = get_reg(c, rn, 0);
   479→        uint32_t offset;
   480→        if (instr & (1u << 25)) {
   481→            shift_result sr = arm_operand2(c, instr);
   482→            offset = sr.value;
   483→        } else {
   484→            offset = instr & 0xFFFu;
   485→        }
   486→        uint32_t addr = up ? base + offset : base - offset;
   487→        if (pre) {
   488→            if (load) {
   489→                if (byte) {
   490→                    c->r[rd] = gba_mem_read8(g, addr);
   491→                } else {
   492→                    uint32_t v = gba_bus_read32(g, addr);
   493→                    uint32_t un = addr & 3u;
   494→                    if (un)
   495→                        v = (v >> (un * 8u)) | (v << (32u - un * 8u));
   496→                    if (rd == 15u)
   497→                        c->r[15] = v & ~3u;
   498→                    else
   499→                        c->r[rd] = v;
   500→                }
   501→            } else if (byte) {
   502→                gba_mem_write8(g, addr, (uint8_t)c->r[rd]);
   503→            } else {
   504→                gba_mem_write32(g, addr, c->r[rd]);
   505→            }
   506→            if (writeback && (!load || rd != rn))
   507→                c->r[rn] = addr;
   508→        } else {
   509→            if (load) {
   510→                if (byte) {
   511→                    c->r[rd] = gba_mem_read8(g, base);
   512→                } else {
   513→                    uint32_t v = gba_bus_read32(g, base);
   514→                    uint32_t un = base & 3u;
   515→                    if (un)
   516→                        v = (v >> (un * 8u)) | (v << (32u - un * 8u));
   517→                    if (rd == 15u)
   518→                        c->r[15] = v & ~3u;
   519→                    else
   520→                        c->r[rd] = v;
   521→                }
   522→            } else if (byte) {
   523→                gba_mem_write8(g, base, (uint8_t)c->r[rd]);
   524→            } else {
   525→                gba_mem_write32(g, base, c->r[rd]);
   526→            }
   527→            c->r[rn] = addr;
   528→        }
   529→        c->r[15] += 4u;
   530→        return load ? 3u : 2u;
   531→    }
   532→
   533→    /* block transfer */
   534→    if ((instr & 0x0E000000u) == 0x08000000u) {
   535→        uint8_t rn = (uint8_t)((instr >> 16) & 0xFu);
   536→        int up = (instr & (1u << 23)) != 0;
   537→        int pre = (instr & (1u << 24)) != 0;
   538→        int writeback = (instr & (1u << 21)) != 0;
   539→        int load = (instr & (1u << 20)) != 0;
   540→        uint32_t list = instr & 0xFFFFu;
   541→        int count = 0;
   542→        for (int i = 0; i < 16; i++)
   543→            if (list & (1u << i))
   544→                count++;
   545→        uint32_t base = c->r[rn];
   546→        uint32_t addr;
   547→        if (up)
   548→            addr = pre ? base + 4u : base;
   549→        else
   550→            addr = base - (uint32_t)count * 4u + (pre ? 0u : 0u);
   551→        uint32_t loaded_pc = 0;
   552→        int has_pc = (list & (1u << 15)) != 0;
   553→        for (int i = 0; i < 16; i++) {
   554→            if (!(list & (1u << i)))
   555→                continue;
   556→            if (load) {
   557→                uint32_t v = gba_bus_read32(g, addr);
   558→                if (i == 15)
   559→                    loaded_pc = v;
   560→                else
   561→                    c->r[i] = v;
   562→            } else {
   563→                uint32_t v = (i == 15) ? c->r[15] + 4u : c->r[i];
   564→                gba_mem_write32(g, addr, v);
   565→            }
   566→            addr += 4u;
   567→        }
   568→        if (load && has_pc) {
   569→            c->cpsr = (c->cpsr & (uint32_t)~GBA_T) |
   570→                      (loaded_pc & 1u ? GBA_T : 0u);
   571→            c->r[15] = loaded_pc & ~3u; /* ARMv4T: no state switch on LDM */
   572→            return (uint32_t)(4u + (uint32_t)count);
   573→        }
   574→        if (writeback && !(load && (list & (1u << rn)) != 0u))
   575→            c->r[rn] = up ? base + (uint32_t)count * 4u
   576→                          : base - (uint32_t)count * 4u;
   577→        c->r[15] += 4u;
   578→        return (uint32_t)(2u + (uint32_t)count);
   579→    }
   580→
   581→    /* branch / branch with link */
   582→    if ((instr & 0x0E000000u) == 0x0A000000u) {
   583→        int32_t off = (int32_t)(instr & 0x00FFFFFFu);
   584→        off = (off << 8) >> 8; /* sign extend */
   585→        int link = (instr & (1u << 24)) != 0;
   586→        if (link)
   587→            c->r[14] = c->r[15]; /* LR = instruction address + 4 */
   588→        c->r[15] = c->r[15] + 4u + ((uint32_t)off << 2u);
   589→        return 3;
   590→    }
   591→
   592→    /* coprocessor space: UND on GBA */
   593→    if ((instr & 0x0C000000u) == 0x0C000000u) {
   594→        gba_cpu_exception(g, 0x04u, GBA_MODE_UND, 4u, 1);
   595→        return 4;
   596→    }
   597→
   598→    /* data processing */
   599→    {
   600→        uint8_t opcode = (uint8_t)((instr >> 21) & 0xFu);
   601→        uint8_t rn = (uint8_t)((instr >> 16) & 0xFu);
   602→        uint8_t rd = (uint8_t)((instr >> 12) & 0xFu);
   603→        int set_flags = (instr & (1u << 20)) != 0;
   604→        uint32_t a = get_reg(c, rn, 4);
   605→        uint32_t b;
   606→        uint8_t shifter_carry = cpsr_c(c);
   607→        if (instr & (1u << 25))
   608→            b = arm_imm(c, instr, &shifter_carry);
   609→        else {
   610→            shift_result sr = arm_operand2(c, instr);
   611→            b = sr.value;
   612→            shifter_carry = sr.carry_out;
   613→        }
   614→        uint32_t result = 0;
   615→        int write = 1;
   616→        switch (opcode) {
   617→        case 0x0: result = a & b; if (set_flags) set_nz32(c, result); break;
   618→        case 0x1: result = a ^ b; if (set_flags) set_nz32(c, result); break;
   619→        case 0x2: result = sub_with_carry(c, a, b, 0, set_flags); break;
   620→        case 0x3: result = sub_with_carry(c, b, a, 0, set_flags); break;
   621→        case 0x4: result = add_with_carry(c, a, b, 0, set_flags); break;
   622→        case 0x5: result = add_with_carry(c, a, b, cpsr_c(c), set_flags); break;
   623→        case 0x6: result = sub_with_carry(c, a, b, 1u - cpsr_c(c), set_flags); break;
   624→        case 0x7: result = sub_with_carry(c, b, a, 1u - cpsr_c(c), set_flags); break;
   625→        case 0x8: result = a & b; write = 0;
   626→                  if (set_flags) set_nz32(c, result); break;
   627→        case 0x9: result = a ^ b; write = 0;
   628→                  if (set_flags) set_nz32(c, result); break;
   629→        case 0xA: result = sub_with_carry(c, a, b, 0, set_flags); write = 0; break;
   630→        case 0xB: result = add_with_carry(c, a, b, 0, set_flags); write = 0; break;
   631→        case 0xC: result = a | b; if (set_flags) set_nz32(c, result); break;
   632→        case 0xD: result = b; if (set_flags) set_nz32(c, result); break;
   633→        case 0xE: result = a & ~b; if (set_flags) set_nz32(c, result); break;
   634→        default: result = ~b; if (set_flags) set_nz32(c, result); break;
   635→        }
   636→        if (set_flags && (opcode <= 1u || opcode == 8u || opcode == 9u ||
   637→                          opcode >= 12u)) {
   638→            /* logic operations: C from the shifter */
   639→            c->cpsr = (c->cpsr & (uint32_t)~GBA_C) |
   640→                      (shifter_carry ? GBA_C : 0u);
   641→        }
   642→        if (write) {
   643→            if (rd == 15u) {
   644→                if (set_flags) {
   645→                    uint32_t spsr = gba_cpu_read_spsr(c);
   646→                    c->cpsr = spsr;
   647→                    gba_cpu_set_mode(c, (uint8_t)(spsr & 0x1Fu));
   648→                    c->r[15] = result;
   649→                } else {
   650→                    c->r[15] = result & ~3u;
   651→                }
   652→                return 4;
   653→            }
   654→            c->r[rd] = result;
   655→        }
   656→        c->r[15] += 4u;
   657→        return 2;
   658→    }
   659→}
   660→
   661→/* ---- Thumb ------------------------------------------------------------------ */
   662→
   663→static uint32_t thumb_step(gba_t *g, gba_cpu *c, uint16_t instr)
   664→{
   665→    uint32_t op = instr >> 11;
   666→
   667→    /* shifted/added: 000 */
   668→    if (op < 3u) {
   669→        uint8_t rd = (uint8_t)(instr & 7u);
   670→        uint8_t rm = (uint8_t)((instr >> 3) & 7u);
   671→        uint32_t amount = (uint32_t)(instr >> 6) & 0x1Fu;
   672→        uint32_t value = c->r[rm];
   673→        shift_result r;
   674→        if (op == 0u && amount == 0u) {
   675→            r.value = value;
   676→            r.carry_out = cpsr_c(c);
   677→        } else {
   678→            r = shift_calc(value, (uint8_t)op, amount, cpsr_c(c));
   679→        }
   680→        c->r[rd] = r.value;
   681→        c->cpsr = (c->cpsr & (uint32_t)~GBA_C) | (r.carry_out ? GBA_C : 0u);
   682→        set_nz32(c, r.value);
   683→        c->r[15] += 2u;
   684→        return 2;
   685→    }
   686→
   687→    /* add/sub: 00011 */
   688→    if (op == 3u) {
   689→        uint8_t rm = (uint8_t)(instr & 7u);
   690→        uint8_t rn = (uint8_t)((instr >> 3) & 7u);
   691→        uint8_t rd = (uint8_t)((instr >> 6) & 7u);
   692→        int imm_form = (instr & (1u << 10)) != 0;
   693→        int is_sub = (instr & (1u << 9)) != 0;
   694→        uint32_t a = c->r[rn];
   695→        uint32_t b = imm_form ? ((instr >> 6) & 7u) : c->r[rm];
   696→        c->r[rd] = is_sub ? sub_with_carry(c, a, b, 0, 1)
   697→                          : add_with_carry(c, a, b, 0, 1);
   698→        c->r[15] += 2u;
   699→        return 2;
   700→    }
   701→
   702→    /* MOV/CMP/ADD/SUB immediate: 001 */
   703→    if (op >= 4u && op <= 7u) {
   704→        uint8_t rd = (uint8_t)((instr >> 8) & 7u);
   705→        uint32_t imm = instr & 0xFFu;
   706→        if (op == 4u) {
   707→            c->r[rd] = imm;
   708→            set_nz32(c, imm);
   709→        } else if (op == 5u) {
   710→            sub_with_carry(c, c->r[rd], imm, 0, 1);
   711→        } else if (op == 6u) {
   712→            c->r[rd] = add_with_carry(c, c->r[rd], imm, 0, 1);
   713→        } else {
   714→            c->r[rd] = sub_with_carry(c, c->r[rd], imm, 0, 1);
   715→        }
   716→        c->r[15] += 2u;
   717→        return 2;
   718→    }
   719→
   720→    /* ALU operations: 010000xxxx */
   721→    if (instr >= 0x4000u && instr < 0x4400u) {
   722→        uint8_t op2 = (uint8_t)((instr >> 6) & 0xFu);
   723→        uint8_t rm = (uint8_t)(instr & 7u);
   724→        uint8_t rd = (uint8_t)((instr >> 3) & 7u);
   725→        uint32_t a = c->r[rd], b = c->r[rm];
   726→        switch (op2) {
   727→        case 0x0: c->r[rd] = a & b; set_nz32(c, c->r[rd]); break;
   728→        case 0x1: c->r[rd] = a ^ b; set_nz32(c, c->r[rd]); break;
   729→        case 0x2: case 0x3: case 0x4: case 0x7: {
   730→            uint8_t type = (uint8_t)(op2 == 0x2u ? 0 : op2 == 0x3u ? 1
   731→                                             : op2 == 0x4u ? 2 : 3);
   732→            shift_result r = shift_calc(a, type, b & 0xFFu, cpsr_c(c));
   733→            c->r[rd] = r.value;
   734→            c->cpsr = (c->cpsr & (uint32_t)~GBA_C) | (r.carry_out ? GBA_C : 0u);
   735→            set_nz32(c, r.value);
   736→            break;
   737→        }
   738→        case 0x5: c->r[rd] = add_with_carry(c, a, b, cpsr_c(c), 1); break;
   739→        case 0x6: c->r[rd] = sub_with_carry(c, a, b, 1u - cpsr_c(c), 1); break;
   740→        case 0x8: set_nz32(c, a & b); break;
   741→        case 0x9: c->r[rd] = sub_with_carry(c, 0, b, 0, 1); break;
   742→        case 0xA: sub_with_carry(c, a, b, 0, 1); break;
   743→        case 0xB: add_with_carry(c, a, b, 0, 1); break;
   744→        case 0xC: c->r[rd] = a | b; set_nz32(c, c->r[rd]); break;
   745→        case 0xD: c->r[rd] = a * b; set_nz32(c, c->r[rd]); break;
   746→        case 0xE: c->r[rd] = a & ~b; set_nz32(c, c->r[rd]); break;
   747→        default: c->r[rd] = ~b; set_nz32(c, c->r[rd]); break;
   748→        }
   749→        c->r[15] += 2u;
   750→        return 2;
   751→    }
   752→
   753→    /* PC-relative load: 01001 */
   754→    if ((instr & 0xF800u) == 0x4800u) {
   755→        uint8_t rd = (uint8_t)((instr >> 8) & 7u);
   756→        uint32_t addr = ((c->r[15] + 4u) & ~3u) + (uint32_t)(instr & 0xFFu) * 4u;
   757→        c->r[rd] = gba_bus_read32(g, addr);
   758→        c->r[15] += 2u;
   759→        return 3;
   760→    }
   761→
   762→    /* Hi register ops / BX: 010001 */
   763→    if ((instr & 0xF000u) == 0x4000u && (instr & 0x0800u) != 0u) {
   764→        uint8_t op2 = (uint8_t)((instr >> 8) & 3u);
   765→        uint8_t rm = (uint8_t)((instr >> 3) & 0xFu);
   766→        uint8_t rd = (uint8_t)((instr & 7u) | ((instr >> 4) & 8u));
   767→        uint32_t b = c->r[rm];
   768→        if (op2 == 3u) { /* BX */
   769→            uint32_t target = c->r[rm];
   770→            c->cpsr = (c->cpsr & (uint32_t)~GBA_T) | (target & 1u ? GBA_T : 0u);
   771→            c->r[15] = target & ~1u;
   772→            return 3;
   773→        }
   774→        if (op2 == 0u) { /* ADD */
   775→            uint32_t r = c->r[rd] + b;
   776→            if (rd == 15u)
   777→                c->r[15] = r & ~1u;
   778→            else
   779→                c->r[rd] = r;
   780→        } else if (op2 == 1u) { /* CMP */
   781→            sub_with_carry(c, c->r[rd], b, 0, 1);
   782→        } else { /* MOV */
   783→            if (rd == 15u)
   784→                c->r[15] = b & ~1u;
   785→            else
   786→                c->r[rd] = b;
   787→        }
   788→        if (rd == 15u && op2 != 1u)
   789→            return 3;
   790→        c->r[15] += 2u;
   791→        return 2;
   792→    }
   793→
   794→    /* load/store with register offset + sign-extended/halfword: 0101 */
   795→    if ((instr & 0xF000u) == 0x5000u) {
   796→        uint8_t rm = (uint8_t)(instr & 7u);
   797→        uint8_t rn = (uint8_t)((instr >> 3) & 7u);
   798→        uint8_t rd = (uint8_t)((instr >> 6) & 7u);
   799→        uint32_t addr = c->r[rn] + c->r[rm];
   800→        uint8_t op2 = (uint8_t)((instr >> 10) & 3u);
   801→        uint8_t kind = (uint8_t)(((instr >> 10) & 1u) | ((instr >> 6) & 2u));
   802→        if ((instr & 0x0200u) == 0u) { /* 0101 0xx: STR/LDRB/LDR/STRB... */
   803→            switch (op2) {
   804→            case 0: gba_mem_write32(g, addr, c->r[rd]); break;
   805→            case 1: gba_mem_write8(g, addr, (uint8_t)c->r[rd]); break;
   806→            case 2: c->r[rd] = gba_bus_read32(g, addr); break;
   807→            default: c->r[rd] = gba_mem_read8(g, addr); break;
   808→            }
   809→            c->r[15] += 2u;
   810→            return (op2 >= 2u) ? 3u : 2u;
   811→        }
   812→        /* 0101 1xx: STRH/LDSB/LDRH/LDSH */
   813→        switch (kind) {
   814→        case 0: gba_mem_write16(g, addr, (uint16_t)c->r[rd]); break;
   815→        case 1: c->r[rd] = (uint32_t)(int32_t)(int8_t)gba_mem_read8(g, addr); break;
   816→        case 2: c->r[rd] = gba_bus_read16(g, addr); break;
   817→        default: c->r[rd] = (uint32_t)(int32_t)(int16_t)gba_bus_read16(g, addr); break;
   818→        }
   819→        c->r[15] += 2u;
   820→        return (kind & 2u) ? 3u : 2u;
   821→    }
   822→
   823→    /* STR/LDRB/LDR/LDRB immediate: 011 */
   824→    if ((instr & 0xE000u) == 0x6000u) {
   825→        uint8_t op2 = (uint8_t)((instr >> 11) & 3u);
   826→        uint32_t imm = (instr >> 6) & 0x1Fu;
   827→        uint8_t rn = (uint8_t)((instr >> 3) & 7u);
   828→        uint8_t rd = (uint8_t)(instr & 7u);
   829→        uint32_t addr = c->r[rn] + (op2 < 2u ? imm : imm * 4u);
   830→        if (op2 == 0u) {
   831→            gba_mem_write32(g, addr, c->r[rd]);
   832→        } else if (op2 == 1u) {
   833→            gba_mem_write8(g, addr, (uint8_t)c->r[rd]);
   834→        } else if (op2 == 2u) {
   835→            c->r[rd] = gba_bus_read32(g, addr);
   836→        } else {
   837→            c->r[rd] = gba_mem_read8(g, addr);
   838→        }
   839→        c->r[15] += 2u;
   840→        return (op2 >= 2u) ? 3u : 2u;
   841→    }
   842→
   843→    /* STRH/LDRH immediate: 1000 */
   844→    if ((instr & 0xF000u) == 0x8000u) {
   845→        uint32_t imm = (uint32_t)((instr >> 6) & 0x1Fu) * 2u;
   846→        uint8_t rn = (uint8_t)((instr >> 3) & 7u);
   847→        uint8_t rd = (uint8_t)(instr & 7u);
   848→        uint32_t addr = c->r[rn] + imm;
   849→        if (instr & (1u << 11))
   850→            c->r[rd] = gba_bus_read16(g, addr);
   851→        else
   852→            gba_mem_write16(g, addr, (uint16_t)c->r[rd]);
   853→        c->r[15] += 2u;
   854→        return (instr & (1u << 11)) ? 3u : 2u;
   855→    }
   856→
   857→    /* STR/LDR SP-relative: 1001 */
   858→    if ((instr & 0xF000u) == 0x9000u) {
   859→        uint8_t rd = (uint8_t)((instr >> 8) & 7u);
   860→        uint32_t addr = c->r[13] + (uint32_t)(instr & 0xFFu) * 4u;
   861→        if (instr & (1u << 11))
   862→            c->r[rd] = gba_bus_read32(g, addr);
   863→        else
   864→            gba_mem_write32(g, addr, c->r[rd]);
   865→        c->r[15] += 2u;
   866→        return (instr & (1u << 11)) ? 3u : 2u;
   867→    }
   868→
   869→    /* load address: 1010 */
   870→    if ((instr & 0xF000u) == 0xA000u) {
   871→        uint8_t rd = (uint8_t)((instr >> 8) & 7u);
   872→        uint32_t imm = (uint32_t)(instr & 0xFFu) * 4u;
   873→        c->r[rd] = (instr & (1u << 11)) ? c->r[13] + imm
   874→                                        : ((c->r[15] + 4u) & ~3u) + imm;
   875→        c->r[15] += 2u;
   876→        return 2;
   877→    }
   878→
   879→    /* push/pop: 1011 x10x */
   880→    if ((instr & 0xF600u) == 0xB400u) {
   881→        int load = (instr & (1u << 11)) != 0;
   882→        int pc_lr = (instr & (1u << 8)) != 0;
   883→        uint32_t count = 0;
   884→        for (int i = 0; i < 8; i++)
   885→            if (instr & (1u << i))
   886→                count++;
   887→        count += (uint32_t)pc_lr;
   888→        uint32_t addr = c->r[13];
   889→        if (!load)
   890→            c->r[13] -= count * 4u;
   891→        else
   892→            addr = c->r[13];
   893→        addr = c->r[13];
   894→        if (!load)
   895→            addr = c->r[13];
   896→        for (int i = 0; i < 8; i++) {
   897→            if (!(instr & (1u << i)))
   898→                continue;
   899→            if (!load) {
   900→                addr += 4u;
   901→                gba_mem_write32(g, addr - 4u, c->r[i]);
   902→            } else {
   903→                c->r[i] = gba_bus_read32(g, addr);
   904→                addr += 4u;
   905→            }
   906→        }
   907→        if (pc_lr) {
   908→            if (load) {
   909→                uint32_t v = gba_bus_read32(g, addr);
   910→                c->r[13] = addr + 4u;
   911→                c->r[15] = v & ~1u;
   912→                return 5;
   913→            }
   914→            gba_mem_write32(g, addr, c->r[14]);
   915→            addr += 4u;
   916→        }
   917→        c->r[13] = load ? addr : c->r[13];
   918→        c->r[15] += 2u;
   919→        return load ? 4u : 3u;
   920→    }
   921→
   922→    /* ADD/SUB SP immediate: 1011 0000x */
   923→    if ((instr & 0xFF00u) == 0xB000u) {
   924→        uint32_t imm = (uint32_t)(instr & 0x7Fu) * 4u;
   925→        c->r[13] = (instr & (1u << 7)) ? c->r[13] + imm : c->r[13] - imm;
   926→        c->r[15] += 2u;
   927→        return 2;
   928→    }
   929→
   930→    /* LDMIA/STMIA: 1100 */
   931→    if ((instr & 0xF000u) == 0xC000u) {
   932→        uint8_t rn = (uint8_t)((instr >> 8) & 7u);
   933→        int load = (instr & (1u << 11)) != 0;
   934→        uint32_t addr = c->r[rn];
   935→        uint32_t base = addr;
   936→        for (int i = 0; i < 8; i++) {
   937→            if (!(instr & (1u << i)))
   938→                continue;
   939→            if (load)
   940→                c->r[i] = gba_bus_read32(g, addr);
   941→            else
   942→                gba_mem_write32(g, addr, c->r[i]);
   943→            addr += 4u;
   944→        }
   945→        if (!(instr & (1u << rn)) || !load)
   946→            c->r[rn] = addr;
   947→        else
   948→            c->r[rn] = base;
   949→        c->r[15] += 2u;
   950→        return load ? 4u : 3u;
   951→    }
   952→
   953→    /* conditional branch: 1101 */
   954→    if ((instr & 0xF000u) == 0xD000u && (instr & 0x0F00u) != 0x0F00u) {
   955→        uint32_t cond = (uint32_t)((instr >> 8) & 0xFu);
   956→        int8_t off = (int8_t)(instr & 0xFFu);
   957→        uint32_t next = c->r[15] + 2u;
   958→        if (cond_ok(cond, c->cpsr)) {
   959→            c->r[15] = next + ((uint32_t)(int32_t)off << 1u);
   960→            return 3;
   961→        }
   962→        c->r[15] += 2u;
   963→        return 2;
   964→    }
   965→
   966→    /* SWI Thumb: 1101 1111 */
   967→    if ((instr & 0xFF00u) == 0xDF00u) {
   968→        gba_swi_hle(g, instr & 0xFFu);
   969→        c->r[15] += 2u;
   970→        return 3;
   971→    }
   972→
   973→    /* unconditional branch: 11100 */
   974→    if ((instr & 0xF800u) == 0xE000u) {
   975→        int32_t off = (int32_t)(instr & 0x7FFu);
   976→        off = (off << 21) >> 21;
   977→        c->r[15] = c->r[15] + 4u + ((uint32_t)off << 1u);
   978→        return 3;
   979→    }
   980→
   981→    /* BL first halfword: 11110 */
   982→    if ((instr & 0xF800u) == 0xF000u) {
   983→        int32_t off = (int32_t)(instr & 0x7FFu);
   984→        off = (off << 21) >> 21;
   985→        c->r[14] = (c->r[15] + 4u + ((uint32_t)off << 12u)) | 1u;
   986→        c->r[15] += 2u;
   987→        return 3;
   988→    }
   989→
   990→    /* BL second halfword: 11111 */
   991→    if ((instr & 0xF800u) == 0xF800u) {
   992→        int32_t off = (int32_t)(instr & 0x7FFu);
   993→        off = (off << 21) >> 21;
   994→        uint32_t target = c->r[14] + ((uint32_t)off << 1u);
   995→        c->r[14] = (c->r[15] + 2u) | 1u;
   996→        c->r[15] = target & ~1u;
   997→        return 4;
   998→    }
   999→
  1000→    /* undefined */
  1001→    c->r[15] += 2u;
  1002→    return 2;
  1003→}
  1004→
  1005→uint32_t gba_cpu_step(gba_t *g)
  1006→{
  1007→    gba_cpu *c = &g->cpu;
  1008→
  1009→    /* returning from an HLE-dispatched IRQ handler: sentinel at BIOS 0 */
  1010→    if (c->bios_dispatch && c->r[15] == 0u) {
  1011→        gba_irq_dispatch_return(g);
  1012→        return 12;
  1013→    }
  1014→
  1015→    if (c->halted) {
  1016→        if ((g->mem.if_reg & g->mem.ie) != 0u)
  1017→            c->halted = 0;
  1018→        return 4;
  1019→    }
  1020→
  1021→    if ((g->mem.if_reg & g->mem.ie) != 0u && (c->cpsr & GBA_I) == 0u &&
  1022→        !c->bios_dispatch) {
  1023→        gba_irq_dispatch(g);
  1024→        return 12;
  1025→    }
  1026→
  1027→    if (c->cpsr & GBA_T) {
  1028→        uint16_t instr = gba_bus_read16(g, c->r[15]);
  1029→        c->r[15] += 2u;
  1030→        return thumb_step(g, c, instr);
  1031→    }
  1032→    uint32_t instr = gba_bus_read32(g, c->r[15]);
  1033→    c->r[15] += 4u;
  1034→    return arm_step(g, c, instr);
  1035→}
